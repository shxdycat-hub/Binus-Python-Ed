def add_numbas(a, b):
    return a + b

def subtract_numbas(a, b):
    return a - b

def multiply_numbas(a, b):
    return a * b

def divide_numbas(a, b):
    if b == 0:
        return None
    return a / b

def modulus_numbas(a, b):
    if b == 0:
        return None
    return a % b
