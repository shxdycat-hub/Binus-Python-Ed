class Student:
    def __init__(self, name: str, ID: str, major: str):
        self.name = name
        self.ID = ID
        self.major = major

    def display_biodata(self):
        print("\n--- Student Biodata ---")
        print(f"Name   : {self.name}")
        print(f"ID     : {self.ID}")
        print(f"Major  : {self.major}")
        print("-----------------------")

if __name__ == "__main__":
    print("=== Enter Student Information ===")
    input_name = input("Enter Name : ")
    input_ID  = input("Enter ID  : ")
    input_major = input("Enter Major: ")

    student_obj = Student(name=input_name, ID=input_ID, major=input_major)
    
    student_obj.display_biodata()
