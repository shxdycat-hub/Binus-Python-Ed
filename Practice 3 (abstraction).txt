from abc import ABC, abstractmethod

class PaymentMethod(ABC):
    @abstractmethod
    def pay(self, amount):
        pass

class CreditCard(PaymentMethod):
    def pay(self, amount):
        print("Paid $%.2f using Credit Card." % amount)
        return True

class EBanking(PaymentMethod):
    def pay(self, amount):
        print("Paid $%.2f using E-Banking." % amount)
        return True 

class BrokenPayment(PaymentMethod):
    pass 

if __name__ == "__main__":
    
    classes_to_test = [CreditCard, EBanking, BrokenPayment]
    payment_methods_list = []
    
    print("Checking and processing configurations:\n")
    
    for i in range(len(classes_to_test)):
        current_class = classes_to_test[i]

        if "pay" in current_class.__dict__ and callable(current_class.pay):
            instance = current_class()
            instance.pay(50.00)
        else:
            print("Process halted at list index %d due to unconfigured class." % i)
            break
