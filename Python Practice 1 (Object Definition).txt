def math_maxxing_normal(a=10, b=3):
    results = {
        "+": a + b,
        "-": a - b,
        "*": a * b,
        "/": None,
        "%": None
    }
    
    if b != 0:
        results["/"] = a / b
        results["%"] = a % b

    return results

    ma_name()

    }
    
    while True:
        menu = input("what operation yo (+|-|/|*|%|stop): ").strip().lower()

        outputs = math_maxxing_normal(n1, n2)
