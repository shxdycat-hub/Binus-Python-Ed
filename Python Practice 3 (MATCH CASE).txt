            match choice:

                case 2:

                    print(f"Result: {temp}°C = {result:.2f} K")
                case 3:
                    result = (temp - 32) * 5/9
                    print(f"Result: {temp}°F = {result:.2f}°C")
                case 4:
                    result = (temp - 32) * 5/9 + 273.15
                    print(f"Result: {temp}°F = {result:.2f} K")
                case 5:
                    if temp < 0:
                        print("Warning: Kelvin cannot be below absolute zero (0 K).")
                    result = temp - 273.15
                    print(f"Result: {temp} K = {result:.2f}°C")
                case 6:
                    if temp < 0:
                        print("Warning: Kelvin cannot be below absolute zero (0 K).")
                    result = (temp - 273.15) * 9/5 + 32
